def write_servo(angle):
  angle = max(0, min(180, angle))
  us = int(500 + (angle / 180.0) * 1900)
  if hasattr(servo_pwm, "duty_ns"):
    servo_pwm.duty_ns(us * 1000)
  else:
    servo_pwm.duty(int((us / 20000.0) * 1023))

# Lop dieu khien OLED SSD1306 128x64 tich hop san (Khong can cai thu vien ngoai)
class SSD1306_I2C(framebuf.FrameBuffer):

  def __init__(self, width, height, i2c, addr=0x3C):
    self.width = width
    self.height = height
    self.i2c = i2c
    self.addr = addr
    self.pages = self.height // 8
    self.buffer = bytearray(self.pages * self.width)
    super().__init__(self.buffer, self.width, self.height, framebuf.MONO_VLSB)
    init_cmds = (
        0xAE,
        0x20,
        0x00,
        0x40,
        0xA1,
        0xA8,
        height - 1,
        0xC8,
        0xD3,
        0x00,
        0xDA,
        0x12,
        0xD5,
        0x80,
        0xD9,
        0xF1,
        0xDB,
        0x30,
        0x81,
        0xFF,
        0xA4,
        0xA6,
        0x8D,
        0x14,
        0xAF,
    )
    for cmd in init_cmds:
      self.write_cmd(cmd)
    self.fill(0)
    self.show()

  def write_cmd(self, cmd):
    self.i2c.writeto(self.addr, bytearray([0x80, cmd]))

  def show(self):
    for cmd in (0x21, 0, self.width - 1, 0x22, 0, self.pages - 1):
      self.write_cmd(cmd)
    self.i2c.writeto(self.addr, b"\x40" + self.buffer)

def relay_set(ch, on):
  if ch < 1 or ch > 4:
    return
  # Khoa cheo an toan giua Bom 1 (Cap) va Bom 2 (Xa)
  if ch == 1 and on and relay_state[1]:
    return
  if ch == 2 and on and relay_state[0]:
    return

  relay_pins[ch - 1].value(RELAY_ON if on else RELAY_OFF)
  relay_state[ch - 1] = bool(on)


def pumps_off():
  relay_set(1, False)
  relay_set(2, False)


def fault(reason):
  global mode, fault_text, status_text, auto_pending
  pumps_off()
  mode = FAULT
  fault_text = reason
  status_text = "LOI: " + reason
  auto_pending = False
  publish_cloud_state()


def stop_cycle():
  global mode, auto_pending, status_text
  pumps_off()
  mode = IDLE
  auto_pending = False
  status_text = "Da dung chu trinh"
  publish_cloud_state()

def load_settings():
  global drain_cm, fill_cm, tank_height_cm, max_minutes
  global feed1, feed2, schedule_enabled, feed_count, last_feed_key, device_id

  try:
    with open(CONFIG_FILE, "r") as f:
      data = json.load(f)
      drain_cm = float(data.get("drain", 12.0))
      fill_cm = float(data.get("fill", 5.0))
      tank_height_cm = float(data.get("height", 20.0))
      max_minutes = int(data.get("maxMin", 5))
      feed1 = str(data.get("feed1", "08:00"))
      feed2 = str(data.get("feed2", "17:00"))
      schedule_enabled = bool(data.get("schedule", True))
      feed_count = int(data.get("feeds", 0))
      last_feed_key = str(data.get("lastKey", ""))
      device_id = str(data.get("devId", "aqua-v3-uet")).strip()
  except Exception:
    pass

  if len(device_id) < 3:
    device_id = "aqua-v3-uet"

  if (
      math.isnan(tank_height_cm)
      or tank_height_cm < 10.0
      or tank_height_cm > 300.0
  ):
    tank_height_cm = 20.0

  if (
      math.isnan(fill_cm)
      or math.isnan(drain_cm)
      or fill_cm < 3.0
      or drain_cm <= fill_cm + 1.0
      or drain_cm > tank_height_cm - 2.0
  ):
    fill_cm = 5.0
    drain_cm = 12.0
    tank_height_cm = 20.0

  if max_minutes < 1 or max_minutes > 30:
    max_minutes = 5

  if feed_count < 0 or feed_count > 5:
    feed_count = 0


def save_settings():
  data = {
      "drain": drain_cm,
      "fill": fill_cm,
      "height": tank_height_cm,
      "maxMin": max_minutes,
      "feed1": feed1,
      "feed2": feed2,
      "schedule": schedule_enabled,
      "feeds": feed_count,
      "lastKey": last_feed_key,
      "devId": device_id,
  }
  try:
    with open(CONFIG_FILE, "w") as f:
      json.dump(data, f)
  except Exception as e:
    print("Loi luu config:", e)


def valid_clock(s):
  if not isinstance(s, str) or len(s) != 5 or s[2] != ":":
    return False
  if not (s[0].isdigit() and s[1].isdigit() and s[3].isdigit() and s[4].isdigit()):
    return False
  h = int(s[0:2])
  m = int(s[3:5])
  return 0 <= h < 24 and 0 <= m < 60

def raw_distance():
  pin_trig.value(0)
  time.sleep_us(3)
  pin_trig.value(1)
  time.sleep_us(10)
  pin_trig.value(0)

  try:
    t = machine.time_pulse_us(pin_echo, 1, 28000)
    if t <= 0:
      return -1.0
    d = (t * 0.0343) / 2.0
    return d if (2.0 <= d <= 400.0) else -1.0
  except Exception:
    return -1.0


def update_distance():
  global last_dist, distance_cm, bad_count, recent_n, recent_pos, valid_count
  if elapsed(last_dist) < 180:
    return
  last_dist = millis()

  d = raw_distance()
  if d < 0 or d > tank_height_cm + 5.0:
    bad_count += 1
    if bad_count >= 3:
      distance_cm = -1.0
      recent_n = 0
      valid_count = 0
    return

  bad_count = 0
  recent[recent_pos] = d
  recent_pos = (recent_pos + 1) % 5
  if recent_n < 5:
    recent_n += 1

  distance_cm = sum(recent[:recent_n]) / recent_n
  if valid_count < 5:
    valid_count += 1


def update_temperature():
  global temp_pending, last_temp_start, temperature_c, ds_roms
  if not ds_roms:
    ds_roms = ds.scan()
    if not ds_roms:
      temperature_c = -127.0
      return

  if not temp_pending:
    try:
      ds.convert_temp()
      last_temp_start = millis()
      temp_pending = True
    except Exception:
      temperature_c = -127.0
      temp_pending = False
  elif elapsed(last_temp_start) >= 750:
    try:
      t = ds.read_temp(ds_roms[0])
      if t is None or t < -50.0 or t > 100.0:
        temperature_c = -127.0
      else:
        temperature_c = float(t)
    except Exception:
      temperature_c = -127.0
    temp_pending = False

def start_feed():
  global servo_busy, servo_phase, servo_at
  if servo_busy or mode == FAULT:
    return
  write_servo(90)
  servo_busy = True
  servo_phase = 1
  servo_at = millis()
  publish_cloud_state()


def update_feed():
  global servo_busy, servo_phase, servo_at, feed_count, auto_pending, status_text
  if not servo_busy:
    return

  if servo_phase == 1 and elapsed(servo_at) >= 1000:
    write_servo(0)
    servo_phase = 2
    servo_at = millis()
  elif servo_phase == 2 and elapsed(servo_at) >= 600:
    servo_busy = False
    servo_phase = 0
    if feed_count < 5:
      feed_count += 1
    save_settings()
    if feed_count >= 5:
      auto_pending = True
      status_text = "Da du 5 lan an - cho thay nuoc"
    publish_cloud_state()


def start_cycle():
  global status_text, cycle_at, phase_at, mode
  if mode != IDLE or servo_busy or valid_count < 5 or distance_cm < 0:
    status_text = "Chua the thay nuoc: kiem tra cam bien"
    return
  if distance_cm >= drain_cm - 0.3:
    status_text = "Muc nuoc da thap - khong xa"
    return
  if distance_cm < 3.0 or distance_cm > tank_height_cm:
    status_text = "Khoang cach khong an toan"
    return

  pumps_off()
  cycle_at = millis()
  phase_at = millis()
  mode = DRAIN
  status_text = "Dang xa nuoc"
  relay_set(2, True)
  publish_cloud_state()


def update_cycle():
  global mode, phase_at, status_text, auto_pending, feed_count
  if mode not in (DRAIN, FILL, SETTLE):
    return

  if valid_count < 5 or distance_cm < 0:
    fault("Mat tin hieu HC-SR04")
    return

  if elapsed(cycle_at) > max_minutes * 60000:
    fault("Qua thoi gian chu trinh")
    return

  if mode == DRAIN:
    if distance_cm >= drain_cm:
      relay_set(2, False)
      mode = SETTLE
      phase_at = millis()
      status_text = "Cho nuoc on dinh"
      publish_cloud_state()

  elif mode == SETTLE:
    if elapsed(phase_at) >= 1500:
      if distance_cm < drain_cm - 0.5:
        fault("Muc nuoc xa chua dat")
        return
      mode = FILL
      phase_at = millis()
      relay_set(1, True)
      status_text = "Dang cap nuoc"
      publish_cloud_state()

  elif mode == FILL:
    if distance_cm <= fill_cm:
      relay_set(1, False)
      mode = IDLE
      status_text = "Thay nuoc hoan tat"
      auto_pending = False
      feed_count = 0
      save_settings()
      publish_cloud_state()

def sync_ntp():
  global clock_valid
  try:
    ntptime.host = "pool.ntp.org"
    ntptime.settime()
    clock_valid = True
  except Exception:
    pass


def update_schedule():
  global last_schedule, clock_valid, last_feed_key
  if elapsed(last_schedule) < 1000:
    return
  last_schedule = millis()

  # Lay gio Viet Nam (UTC + 7h = +25200 giay)
  t = time.localtime(time.time() + 7 * 3600)
  year, mon, mday, hour, minute = t[0], t[1], t[2], t[3], t[4]
  clock_valid = year >= 2025

  if (
      not clock_valid
      or not schedule_enabled
      or servo_busy
      or mode != IDLE
      or auto_pending
  ):
    return

  hhmm = "{:02d}:{:02d}".format(hour, minute)
  key = "{:04d}{:02d}{:02d}{:02d}{:02d}".format(year, mon, mday, hour, minute)

  if (feed1 == hhmm or feed2 == hhmm) and last_feed_key != key:
    last_feed_key = key
    save_settings()
    start_feed()

def update_oled():

  last_oled = millis()

  try:

    t_str = (
        "{:.1f} C".format(temperature_c) if temperature_c > -100 else "ERR C"
    )

    oled.text("Feed: {}/5".format(feed_count), 0, 33)

        "R1:{} R2:{} R3:{}".format(
            int(relay_state[0]), int(relay_state[1]), int(relay_state[2])
        ),
        0,

    ip_str = wlan.ifconfig()[0] if wlan.isconnected() else "WiFi disconnected"
    oled.text(ip_str, 0, 55)
    oled.show()
  except Exception:
    pass

def apply_config_params(a, b, d, f, h, m, sched, new_dev_id):
  global feed1, feed2, drain_cm, fill_cm, tank_height_cm, max_minutes
  global schedule_enabled, device_id, mqtt_connected, mqtt_sock

  if mode != IDLE:
    return False, "Dung chu trinh truoc khi doi nguong"

  if (
      not valid_clock(a)
      or not valid_clock(b)
      or math.isnan(d)
      or math.isnan(f)
      or math.isnan(h)
      or h < 10.0
      or h > 300.0
      or f < 3.0
      or d <= f + 1.0
      or d > h - 2.0
      or m < 1
      or m > 30
  ):
    return False, "Cau hinh khong hop le: can 3 <= cap < xa < chieu cao be"

  feed1 = a
  feed2 = b
  drain_cm = d
  fill_cm = f
  tank_height_cm = h
  max_minutes = m
  schedule_enabled = sched

  new_dev_id = new_dev_id.strip()
  dev_id_changed = False
  if len(new_dev_id) >= 3 and new_dev_id != device_id:
    device_id = new_dev_id
    dev_id_changed = True

  save_settings()
  if dev_id_changed and mqtt_sock:
    try:
      mqtt_sock.close()
    except Exception:
      pass
    mqtt_sock = None
    mqtt_connected = False
  else:
    publish_cloud_config()
    publish_cloud_state()

  return True, "Da luu cau hinh thanh cong"

def mqtt_encode_len(length):
  enc = bytearray()
  while True:
    b = length % 128
    length //= 128
    if length > 0:
      b |= 128
    enc.append(b)
    if length <= 0:
      break
  return enc


def mqtt_publish_raw(topic, payload, retain=False):
  global mqtt_connected, mqtt_sock
  if not mqtt_connected or not mqtt_sock:
    return False
  try:
    t_bytes = topic.encode("utf-8")
    p_bytes = payload.encode("utf-8")
    rem_len = 2 + len(t_bytes) + len(p_bytes)
    hdr = bytearray([0x31 if retain else 0x30]) + mqtt_encode_len(rem_len)
    tl = bytearray([(len(t_bytes) >> 8) & 0xFF, len(t_bytes) & 0xFF])
    mqtt_sock.send(hdr + tl + t_bytes + p_bytes)
    return True
  except Exception:
    mqtt_connected = False
    return False


def mqtt_subscribe_raw(topic):
  global mqtt_connected, mqtt_sock
  if not mqtt_connected or not mqtt_sock:
    return False
  try:
    t_bytes = topic.encode("utf-8")
    rem_len = 2 + 2 + len(t_bytes) + 1
    hdr = bytearray([0x82]) + mqtt_encode_len(rem_len)
    pkt_id = b"\x00\x01"
    tl = bytearray([(len(t_bytes) >> 8) & 0xFF, len(t_bytes) & 0xFF])
    mqtt_sock.send(hdr + pkt_id + tl + t_bytes + b"\x00")
    return True
  except Exception:
    mqtt_connected = False
    return False


def publish_cloud_state():
  if mqtt_connected:
    mqtt_publish_raw(
        "aquasmart_v3/" + device_id + "/data", build_data_json(), True
    )


def publish_cloud_config():
  if mqtt_connected:
    mqtt_publish_raw(
        "aquasmart_v3/" + device_id + "/config", build_config_json(), True
    )


def publish_cloud_toast(msg, ok=True):
  if mqtt_connected:
    payload = json.dumps({"ok": bool(ok), "message": msg})
    mqtt_publish_raw("aquasmart_v3/" + device_id + "/resp", payload, False)


def url_decode(s):
  s = s.replace("+", " ").replace("%3A", ":").replace("%3a", ":")
  return s


def parse_qs(qs):
  res = {}
  if not qs:
    return res
  for pair in qs.split("&"):
    if "=" in pair:
      k, v = pair.split("=", 1)
      res[url_decode(k)] = url_decode(v)
  return res


def handle_cloud_command(cmd):
  cmd = cmd.strip()
  if cmd == "SYNC":
    publish_cloud_config()
    publish_cloud_state()
  elif cmd == "FEED":
    if servo_busy or mode == FAULT or auto_pending or feed_count >= 5:
      publish_cloud_toast("Dang ban hoac cho thay nuoc", False)
      return
    start_feed()
    publish_cloud_toast("Da bat dau cho an", True)
  elif cmd == "WATER_START":
    start_cycle()
    if mode == DRAIN:
      publish_cloud_toast("Da bat dau xa nuoc", True)
    else:
      publish_cloud_toast(status_text, False)
  elif cmd == "WATER_STOP":
    stop_cycle()
    publish_cloud_toast("Da dung chu trinh", True)
  elif cmd.startswith("RELAY:"):
    try:
      ch = int(cmd[6:])
    except Exception:
      ch = 0
    if ch not in (3, 4):
      publish_cloud_toast("Chi dieu khien duoc relay 3/4", False)
      return
    relay_set(ch, not relay_state[ch - 1])
    publish_cloud_state()
    publish_cloud_toast("Da doi trang thai relay", True)
  elif cmd.startswith("CFG:"):
    params = parse_qs(cmd[4:])
    try:
      ok, msg = apply_config_params(
          params.get("feed1", feed1),
          params.get("feed2", feed2),
          float(params.get("drain", drain_cm)),
          float(params.get("fill", fill_cm)),
          float(params.get("height", tank_height_cm)),
          int(params.get("maxMinutes", max_minutes)),
          params.get("schedule", "0") == "1",
          params.get("devId", device_id),
      )
      publish_cloud_toast(msg, ok)
    except Exception:
      publish_cloud_toast("Loi tham so cau hinh", False)


def connect_cloud_mqtt():
  global mqtt_sock, mqtt_connected, last_mqtt_ping
  try:
    if mqtt_sock:
      try:
        mqtt_sock.close()
      except Exception:
        pass
    addr = socket.getaddrinfo(MQTT_HOST, MQTT_PORT)[0][-1]
    s = socket.socket()
    s.settimeout(1.5)
    s.connect(addr)

    cid = ("esp32-" + device_id + "-" + str(millis() % 9000 + 1000)).encode(
        "utf-8"
    )
    rem_len = 10 + 2 + len(cid)
    pkt = (
        bytearray([0x10])
        + mqtt_encode_len(rem_len)
        + b"\x00\x04MQTT\x04\x02\x00\x3c"
        + bytearray([(len(cid) >> 8) & 0xFF, len(cid) & 0xFF])
        + cid
    )
    s.send(pkt)

    ack = s.recv(4)
    if not ack or len(ack) < 4 or ack[0] != 0x20 or ack[3] != 0x00:
      s.close()
      return False

    s.setblocking(False)
    mqtt_sock = s
    mqtt_connected = True

def update_cloud_mqtt():
  global mqtt_connected, last_mqtt_retry, last_mqtt_ping, last_mqtt_pub
  if not wlan.isconnected():
    mqtt_connected = False
    return

  if not mqtt_connected or not mqtt_sock:
    if mode == IDLE and not servo_busy and elapsed(last_mqtt_retry) >= 8000:
      last_mqtt_retry = millis()
      connect_cloud_mqtt()
    return

  try:
    data = mqtt_sock.recv(512)
    if data:
      idx = 0
      n = len(data)
      while idx < n:
        hdr = data[idx]
        idx += 1
        mult = 1
        rem_len = 0
        while idx < n:
          enc = data[idx]
          idx += 1
          rem_len += (enc & 127) * mult
          mult *= 128
          if (enc & 128) == 0:
            break
        if idx + rem_len <= n and (hdr & 0xF0) == 0x30 and rem_len >= 2:
          t_len = (data[idx] << 8) | data[idx + 1]
          offset = idx + 2 + t_len
          if (hdr & 0x06) != 0:
            offset += 2
          end_pkt = idx + rem_len
          if offset <= end_pkt:
            payload = data[offset:end_pkt].decode("utf-8", "ignore")
            handle_cloud_command(payload)
        idx += rem_len
    elif data == b"":
      mqtt_connected = False
      return
  except OSError:
    pass
  except Exception:
    mqtt_connected = False
    return

  if elapsed(last_mqtt_ping) >= 25000:
    last_mqtt_ping = millis()
    try:
      mqtt_sock.send(b"\xc0\x00")
    except Exception:
      mqtt_connected = False
      return

  if elapsed(last_mqtt_pub) >= 1200:
    last_mqtt_pub = millis()
    publish_cloud_state()

http_sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
http_sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
http_sock.bind(("0.0.0.0", 80))
http_sock.listen(3)

poller = select.poll()
poller.register(http_sock, select.POLLIN)


def send_http_json(conn, body_str, status="200 OK"):
  hdr = (
      "HTTP/1.1 "
      + status
      + "\r\n"
      "Content-Type: application/json; charset=utf-8\r\n"
      "Access-Control-Allow-Origin: *\r\n"
      "Access-Control-Allow-Methods: GET, POST, OPTIONS\r\n"
      "Access-Control-Allow-Headers: Content-Type\r\n"
      "Connection: close\r\n\r\n"
  )
  conn.sendall(hdr.encode("utf-8") + body_str.encode("utf-8"))


def send_msg(conn, msg, code=200):
  status_map = {
      200: "200 OK",
      400: "400 Bad Request",
      409: "409 Conflict",
      404: "404 Not Found",
  }
  send_http_json(
      conn, json.dumps({"message": msg}), status_map.get(code, "200 OK")
  )


def handle_http_client():
  events = poller.poll(0)
  if not events:
    return
  conn = None
  try:
    conn, _ = http_sock.accept()
    conn.settimeout(1.5)
    req = conn.recv(1024).decode("utf-8", "ignore")
    if not req:
      conn.close()
      return

    lines = req.split("\r\n")
    parts = lines[0].split(" ")
    if len(parts) < 2:
      conn.close()
      return
    method, full_path = parts[0], parts[1]

    if method == "OPTIONS":
      send_http_json(conn, "{}")
      conn.close()
      return

    path = full_path.split("?")[0]
    query_str = full_path.split("?")[1] if "?" in full_path else ""
    body_str = req.split("\r\n\r\n", 1)[1] if "\r\n\r\n" in req else ""

    if path == "/" and method == "GET":
      conn.sendall(
          b"HTTP/1.1 200 OK\r\nContent-Type: text/html;"
          b" charset=utf-8\r\nAccess-Control-Allow-Origin: *\r\nConnection:"
          b" close\r\n\r\n"
      )
      mv = memoryview(PAGE_HTML.encode("utf-8"))
      for i in range(0, len(mv), 1024):
        conn.sendall(mv[i : i + 1024])

    elif path == "/data" and method == "GET":
      send_http_json(conn, build_data_json())

    elif path == "/config" and method == "GET":
      send_http_json(conn, build_config_json())

    elif path == "/config" and method == "POST":
      params = parse_qs(body_str if body_str else query_str)
      req_keys = (
          "feed1",
          "feed2",
          "drain",
          "fill",
          "height",
          "maxMinutes",
          "schedule",
      )
      if not all(k in params for k in req_keys):
        send_msg(conn, "Thieu tham so", 400)
      else:
        ok, msg = apply_config_params(
            params["feed1"],
            params["feed2"],
            float(params["drain"]),
            float(params["fill"]),
            float(params["height"]),
            int(params["maxMinutes"]),
            params["schedule"] == "1",
            params.get("devId", device_id),
        )
        send_msg(conn, msg, 200 if ok else (409 if mode != IDLE else 400))

    elif path == "/feed" and method == "POST":
      if servo_busy or mode == FAULT or auto_pending or feed_count >= 5:
        send_msg(conn, "Dang ban hoac cho thay nuoc", 409)
      else:
        start_feed()
        send_msg(conn, "Da bat dau cho an", 200)

    elif path == "/water/start" and method == "POST":
      start_cycle()
      if mode == DRAIN:
        send_msg(conn, "Da bat dau xa nuoc", 200)
      else:
        send_msg(conn, status_text, 409)

    elif path == "/water/stop" and method == "POST":
      stop_cycle()
      send_msg(conn, "Da dung chu trinh", 200)

    elif path == "/relay" and method == "POST":
      params = parse_qs(query_str if query_str else body_str)
      ch = int(params.get("ch", "0"))
      if ch not in (3, 4):
        send_msg(conn, "Chi dieu khien duoc relay 3/4", 400)
      else:
        relay_set(ch, not relay_state[ch - 1])
        publish_cloud_state()
        send_msg(conn, "Da doi trang thai relay", 200)
    else:
      send_msg(conn, "Not Found", 404)

  except Exception:
    pass
  finally:
    if conn:
      try:
        conn.close()
      except Exception:
        pass
    gc.collect()
