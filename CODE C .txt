void relaySet(int ch, bool on) {
  if (ch < 1 || ch > 4) return;


  digitalWrite(
    relayPins[ch - 1],
    on ? RELAY_ON : RELAY_OFF
  );


  relayState[ch - 1] = on;
}


void pumpsOff() {
  relaySet(1, false);
  relaySet(2, false);
}


void fault(String reason) {
  pumpsOff();
  mode = FAULT;
  faultText = reason;
  statusText = "LOI: " + reason;
  autoPending = false;
  publishCloudState();
}


void stopCycle() {
  pumpsOff();
  mode = IDLE;
  autoPending = false;
  statusText = "Da dung chu trinh";
  publishCloudState();
}

// =====================================
// TIME VALIDATION
// =====================================


bool validClock(String s) {
  if (s.length() != 5 || s[2] != ':') return false;


  for (int i : {0, 1, 3, 4}) {
    if (s[i] < '0' || s[i] > '9') return false;
  }


  int h = s.substring(0, 2).toInt();
  int m = s.substring(3, 5).toInt();


  return h < 24 && m < 60;
}


// =====================================
// HC-SR04
// =====================================


float rawDistance() {
  digitalWrite(5, LOW);
  delayMicroseconds(3);


  digitalWrite(5, HIGH);
  delayMicroseconds(10);


  digitalWrite(5, LOW);


  unsigned long t = pulseIn(18, HIGH, 28000UL);


  if (t == 0) return -1;


  float d = t * 0.0343f / 2.0f;


  return (d >= 2 && d <= 400) ? d : -1;
}


void updateDistance() {
  if (millis() - lastDist < 180) return;


  lastDist = millis();


  float d = rawDistance();


  if (d < 0 || d > tankHeightCm + 5) {
    badCount++;


    if (badCount >= 3) {
      distanceCm = -1;
      recentN = 0;
      validCount = 0;
    }


    return;
  }


  badCount = 0;


  recent[recentPos] = d;
  recentPos = (recentPos + 1) % 5;


  if (recentN < 5) recentN++;


  float s = 0;


  for (int i = 0; i < recentN; i++) {
    s += recent[i];
  }


  distanceCm = s / recentN;


  if (validCount < 5) validCount++;
}

void updateTemperature() {
  if (!tempPending) {
    ds.requestTemperatures();
    lastTempStart = millis();
    tempPending = true;
  }
  else if (millis() - lastTempStart >= 750) {
    float t = ds.getTempCByIndex(0);


    temperatureC = (t == DEVICE_DISCONNECTED_C || t < -50 || t > 100) ? -127 : t;


    tempPending = false;
  }
}


// =====================================
// START WATER CYCLE
// =====================================


void startCycle() {
  if (mode != IDLE || servoBusy || validCount < 5 || distanceCm < 0) {
    statusText = "Chua the thay nuoc: kiem tra cam bien";
    return;
  }


  pumpsOff();


  cycleAt = millis();
  phaseAt = millis();


  mode = DRAIN;
  statusText = "Dang xa nuoc";
  autoPending = false;


  relaySet(2, true);
  publishCloudState();
}


// =====================================
// SERVO FEEDING
// =====================================


void startFeed() {
  if (servoBusy || mode == FAULT) return;


  feeder.write(90);


  servoBusy = true;
  servoPhase = 1;
  servoAt = millis();
  publishCloudState();
}


void updateFeed() {
  if (!servoBusy) return;


  if (servoPhase == 1 && millis() - servoAt >= 1000) {
    feeder.write(0);


    servoPhase = 2;
    servoAt = millis();
  }
  else if (servoPhase == 2 && millis() - servoAt >= 600) {
    servoBusy = false;
    servoPhase = 0;


    feedCount++;
   
    if (feedCount >= maxFeeds) {
      feedCount = 0;
      autoPending = true;
      statusText = "Da du " + String(maxFeeds) + " lan an - tu dong thay nuoc";
      startCycle();
    }


    prefs.putInt("feeds", feedCount);
    publishCloudState();
  }
}


void updateAutoWater() {
  if (autoPending && mode == IDLE && !servoBusy) {
    startCycle();
  }
}


// =====================================
// WATER CYCLE
// =====================================


void updateCycle() {
  if (mode != DRAIN && mode != FILL && mode != SETTLE) return;


  if (validCount < 5 || distanceCm < 0) {
    fault("Mat tin hieu HC-SR04");
    return;
  }


  if (millis() - cycleAt > (unsigned long)maxMinutes * 60000UL) {
    fault("Qua thoi gian chu trinh");
    return;
  }


  if (mode == DRAIN) {
    if (distanceCm >= drainCm) {
      relaySet(2, false);


      mode = SETTLE;
      phaseAt = millis();


      statusText = "Cho nuoc on dinh";
      publishCloudState();
    }
  }
  else if (mode == SETTLE) {
    if (millis() - phaseAt >= 1500) {
      mode = FILL;
      phaseAt = millis();


      relaySet(1, true);


      statusText = "Dang cap nuoc";
      publishCloudState();
    }
  }
  else if (mode == FILL) {
    if (distanceCm <= fillCm) {
      relaySet(1, false);


      mode = IDLE;
      statusText = "Thay nuoc hoan tat";


      autoPending = false;
      feedCount = 0;


      prefs.putInt("feeds", feedCount);
      publishCloudState();
    }
  }
}


// =====================================
// AUTOMATIC FEEDING SCHEDULE
// =====================================


void updateSchedule() {
  if (millis() - lastSchedule < 1000) return;


  lastSchedule = millis();


  struct tm t;


  clockValid = getLocalTime(&t, 0) && t.tm_year + 1900 >= 2025;


  if (!clockValid || !scheduleEnabled || servoBusy || mode != IDLE || autoPending) return;


  char hhmm[6];
  char key[20];


  snprintf(hhmm, sizeof(hhmm), "%02d:%02d", t.tm_hour, t.tm_min);
  snprintf(key, sizeof(key), "%04d%02d%02d%02d%02d", t.tm_year + 1900, t.tm_mon + 1, t.tm_mday, t.tm_hour, t.tm_min);


  if ((feed1 == hhmm || feed2 == hhmm) && lastFeedKey != key) {
    lastFeedKey = key;
    prefs.putString("lastKey", lastFeedKey);
    startFeed();
  }
}

void handleRelay() {
  if (!server.hasArg("ch")) {
    sendMsg("Thieu kenh relay", 400);
    return;
  }
  int ch = server.arg("ch").toInt();
  if (ch < 1 || ch > 4) {
    sendMsg("Kenh relay khong hop le", 400);

  publishCloudState();
  sendMsg("Da doi trang thai relay " + String(ch), 200);
}

// =====================================


bool applyConfigParams(String a, String b, float d, float f, float h, int m, int mf, bool sched, String newDevId, String &outMsg) {
  if (mode != IDLE) {
    outMsg = "Dung chu trinh truoc khi doi nguong";
    return false;
  }


  if (!validClock(a) || !validClock(b) || !isfinite(d) || !isfinite(f) || !isfinite(h) || m < 1 || mf < 1) {
    outMsg = "Cau hinh khong hop le!";
    return false;
  }


  feed1 = a;
  feed2 = b;
  drainCm = d;
  fillCm = f;
  tankHeightCm = h;
  maxMinutes = m;
  maxFeeds = mf;
  scheduleEnabled = sched;


  newDevId.trim();
  bool devIdChanged = false;
  if (newDevId.length() >= 3 && newDevId != deviceId) {
    deviceId = newDevId;
    devIdChanged = true;
  }


  saveSettings();
  if (devIdChanged && mqttNet.connected()) {
    mqttNet.stop();
    mqttConnected = false;
  } else {
    publishCloudConfig();
    publishCloudState();
  }


  outMsg = "Da luu cau hinh thanh cong";
  return true;
}


void saveConfig() {
  if (!server.hasArg("feed1") || !server.hasArg("feed2") || !server.hasArg("drain") ||
      !server.hasArg("fill") || !server.hasArg("height") || !server.hasArg("maxMinutes") ||
      !server.hasArg("maxFeeds") || !server.hasArg("schedule")) {
    sendMsg("Thieu tham so", 400);
    return;
  }


  String msg;
  String newId = server.hasArg("devId") ? server.arg("devId") : deviceId;
  bool ok = applyConfigParams(
    server.arg("feed1"), server.arg("feed2"),
    server.arg("drain").toFloat(), server.arg("fill").toFloat(),
    server.arg("height").toFloat(), server.arg("maxMinutes").toInt(),
    server.arg("maxFeeds").toInt(),
    server.arg("schedule") == "1", newId, msg
  );


  sendMsg(msg, ok ? 200 : (mode != IDLE ? 409 : 400));
}


// =====================================
// GIAO THUC CLOUD MQTT
// =====================================


void mqttWriteRemLen(uint8_t* buf, int &pos, int len) {
  do {
    uint8_t b = len % 128;
    len /= 128;
    if (len > 0) b |= 128;
    buf[pos++] = b;
  } while (len > 0);
}


bool mqttPublishRaw(String topic, String payload, bool retain = false) {
  if (!mqttNet.connected()) return false;
  int tLen = topic.length();
  int pLen = payload.length();
  int remLen = 2 + tLen + pLen;


  uint8_t hdr[6];
  int hPos = 0;
  hdr[hPos++] = retain ? 0x31 : 0x30;
  mqttWriteRemLen(hdr, hPos, remLen);


  mqttNet.write(hdr, hPos);
  uint8_t tl[2] = {(uint8_t)(tLen >> 8), (uint8_t)(tLen & 0xFF)};
  mqttNet.write(tl, 2);
  mqttNet.print(topic);
  mqttNet.print(payload);
  return true;
}


bool mqttSubscribeRaw(String topic) {
  if (!mqttNet.connected()) return false;
  int tLen = topic.length();
  int remLen = 2 + 2 + tLen + 1;


  uint8_t hdr[6];
  int hPos = 0;
  hdr[hPos++] = 0x82;
  mqttWriteRemLen(hdr, hPos, remLen);


  mqttNet.write(hdr, hPos);
  uint8_t pktId[2] = {0x00, 0x01};
  mqttNet.write(pktId, 2);
  uint8_t tl[2] = {(uint8_t)(tLen >> 8), (uint8_t)(tLen & 0xFF)};
  mqttNet.write(tl, 2);
  mqttNet.print(topic);
  uint8_t qos = 0x00;
  mqttNet.write(&qos, 1);
  return true;
}

  if (!mqttConnected) return;

  mqttPublishRaw("aquasmart_v3/" + deviceId + "/resp", j, false);

String getQueryVal(String qs, String key) {
  String k = key + "=";
  int idx = qs.indexOf(k);
  while (idx != -1) {
    if (idx == 0 || qs[idx - 1] == '&') {
      int endIdx = qs.indexOf('&', idx + k.length());
      if (endIdx == -1) return qs.substring(idx + k.length());
      return qs.substring(idx + k.length(), endIdx);
    }
    idx = qs.indexOf(k, idx + 1);
  }
  return "";
}


void handleCloudCommand(String cmd) {
  cmd.trim();
  if (cmd == "SYNC") {
    publishCloudConfig();
    publishCloudState();
  }
  else if (cmd == "FEED") {
    if (servoBusy || mode == FAULT || autoPending) {
      publishCloudToast("Dang ban hoac cho thay nuoc", false);
      return;
    }
    startFeed();
    publishCloudToast("Da bat dau cho an", true);
  }
  else if (cmd == "WATER_START") {
    startCycle();
    if (mode == DRAIN) {
      publishCloudToast("Da bat dau xa nuoc", true);
    } else {
      publishCloudToast(statusText, false);
    }
  }
  else if (cmd == "WATER_STOP") {
    stopCycle();
    publishCloudToast("Da dung chu trinh", true);
  }
  else if (cmd.startsWith("RELAY:")) {
    int ch = cmd.substring(6).toInt();
    if (ch < 1 || ch > 4) {
      publishCloudToast("Kenh relay khong hop le", false);
      return;
    }
    relaySet(ch, !relayState[ch - 1]);
    publishCloudState();
    publishCloudToast("Da doi trang thai relay " + String(ch), true);
  }
  else if (cmd.startsWith("CFG:")) {
    String qs = cmd.substring(4);
    String msg;
    String newId = getQueryVal(qs, "devId");
    if (newId == "") newId = deviceId;
    bool ok = applyConfigParams(
      getQueryVal(qs, "feed1"), getQueryVal(qs, "feed2"),
      getQueryVal(qs, "drain").toFloat(), getQueryVal(qs, "fill").toFloat(),
      getQueryVal(qs, "height").toFloat(), getQueryVal(qs, "maxMinutes").toInt(),
      getQueryVal(qs, "maxFeeds").toInt(),
      getQueryVal(qs, "schedule") == "1", newId, msg
    );
    publishCloudToast(msg, ok);
  }
}


bool connectCloudMqtt() {
  if (!mqttNet.connect(MQTT_HOST, MQTT_PORT, 1500)) return false;


  String cid = "esp32-" + deviceId + "-" + String(random(1000, 9999));
  int cLen = cid.length();
  int remLen = 10 + 2 + cLen;


  uint8_t pkt[64];
  int pos = 0;
  pkt[pos++] = 0x10;
  mqttWriteRemLen(pkt, pos, remLen);
  pkt[pos++] = 0x00; pkt[pos++] = 0x04;
  pkt[pos++] = 'M';  pkt[pos++] = 'Q'; pkt[pos++] = 'T'; pkt[pos++] = 'T';
  pkt[pos++] = 0x04;
  pkt[pos++] = 0x02;
  pkt[pos++] = 0x00; pkt[pos++] = 0x3C;
  pkt[pos++] = (uint8_t)(cLen >> 8);
  pkt[pos++] = (uint8_t)(cLen & 0xFF);


  mqttNet.write(pkt, pos);
  mqttNet.print(cid);


  unsigned long t0 = millis();
  while (mqttNet.available() < 4) {
    if (millis() - t0 > 1200) {
      mqttNet.stop();
      return false;
    }
    delay(5);
  }


  uint8_t ack[4];
  mqttNet.read(ack, 4);
  if (ack[0] != 0x20 || ack[3] != 0x00) {
    mqttNet.stop();
    return false;
  }


  mqttConnected = true;
  lastMqttPing = millis();
  mqttSubscribeRaw("aquasmart_v3/" + deviceId + "/cmd");
  publishCloudConfig();
  publishCloudState();
  return true;
}


void updateCloudMqtt() {
  if (WiFi.status() != WL_CONNECTED) {
    mqttConnected = false;
    return;
  }


  if (!mqttNet.connected()) {
    mqttConnected = false;
    if (mode == IDLE && !servoBusy && millis() - lastMqttRetry >= 8000) {
      lastMqttRetry = millis();
      connectCloudMqtt();
    }
    return;
  }


  while (mqttNet.available() > 0) {
    uint8_t hdr = mqttNet.read();
    int mult = 1;
    int remLen = 0;
    unsigned long t0 = millis();
    uint8_t enc;
    do {
      while (!mqttNet.available()) {
        if (millis() - t0 > 300) return;
      }
      enc = mqttNet.read();
      remLen += (enc & 127) * mult;
      mult *= 128;
    } while ((enc & 128) != 0);


    if (remLen <= 0) continue;
    if (remLen > 512) {
      while (remLen-- > 0 && mqttNet.available()) mqttNet.read();
      continue;
    }


    uint8_t buf[513];
    int nRead = 0;
    t0 = millis();
    while (nRead < remLen && millis() - t0 < 400) {
      if (mqttNet.available()) buf[nRead++] = mqttNet.read();
    }
    if (nRead < remLen) continue;


    if ((hdr & 0xF0) == 0x30 && remLen >= 2) {
      int tLen = (buf[0] << 8) | buf[1];
      int offset = 2 + tLen;
      if ((hdr & 0x06) != 0) offset += 2;
      if (offset <= remLen) {
        String payload = "";
        for (int i = offset; i < remLen; i++) payload += (char)buf[i];
        handleCloudCommand(payload);
      }
    }
  }


  if (millis() - lastMqttPing >= 25000) {
    lastMqttPing = millis();
    uint8_t ping[2] = {0xC0, 0x00};
    mqttNet.write(ping, 2);
  }


  if (millis() - lastMqttPub >= 1200) {
    lastMqttPub = millis();
    publishCloudState();
  }
}

void setup() {
  Serial.begin(115200);
  delay(500);


  for (int i = 0; i < 4; i++) {
    pinMode(relayPins[i], OUTPUT);
    digitalWrite(relayPins[i], RELAY_OFF);
    relayState[i] = false;
  }


  pinMode(5, OUTPUT);   // Trig
  pinMode(18, INPUT);   // Echo


  ds.begin();
  feeder.attach(13);
  feeder.write(0);


  loadSettings();


  WiFi.begin(ssid, password);
  Serial.print("Dang ket noi WiFi");
  unsigned long startAttempt = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - startAttempt < 15000) {
    delay(500);
    Serial.print(".");
  }


  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nWiFi connected! IP: " + WiFi.localIP().toString());
    configTime(25200, 0, "pool.ntp.org", "time.nist.gov");
  } else {
    Serial.println("\nWiFi ket noi that bai, chay che do offline.");
  }


  Wire.begin(21, 22);
  if (oled.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    oledReady = true;
    oled.clearDisplay();
    oled.setTextSize(1);
    oled.setTextColor(SSD1306_WHITE);
    oled.setCursor(0, 0);
    oled.println("AQUA SMART V3.0");
    oled.println("Khoi dong thanh cong!");
    oled.display();
    delay(1000);
  }


  server.on("/", []() {
    server.send_P(200, "text/html", PAGE);
  });
  server.on("/data", sendData);
  server.on("/config", HTTP_GET, []() {
    if (server.hasArg("feed1")) saveConfig();
    else getConfig();
  });
  server.on("/feed", []() {
    if (servoBusy || mode == FAULT || autoPending) {
      sendMsg("Dang ban hoac cho thay nuoc", 409);
      return;
    }
    startFeed();
    sendMsg("Da bat dau cho an", 200);
  });
  server.on("/water/start", []() {
    startCycle();
    sendMsg(statusText, mode == DRAIN ? 200 : 400);
  });
  server.on("/water/stop", []() {
    stopCycle();
    sendMsg("Da dung chu trinh", 200);
  });
  server.on("/relay", handleRelay);


  server.begin();
  Serial.println("Web server da khoi dong!");
}

  server.handleClient();
  updateDistance();
  updateTemperature();
  updateFeed();
  updateAutoWater();
  updateCycle();
  updateSchedule();
  updateCloudMqtt();
 
  if (oledReady) {
    updateOLED();
  }
}
